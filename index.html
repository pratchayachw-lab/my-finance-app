        function handleTransaction(e) {

            e.preventDefault();

            if (editingTxId) revertBalance(editingTxId);


            const type = document.getElementById('tx-type')?.value;

            const amountStr = document.getElementById('tx-amount')?.value;

            const fromId = parseInt(document.getElementById('from-account')?.value);

            

            if (!type || !amountStr || isNaN(fromId)) return;

            

            const amount = parseFloat(amountStr);

            const fromAcc = accounts.find(a => a.id === fromId);


            if ((type === 'expense' || type === 'transfer') && amount > fromAcc.balance && fromAcc.type !== 'debt') {

                return alert("ยอดเงินไม่พอ! กรุณายืมจากหนี้สินหรือโอนเงินเข้ามาก่อน");

            }


            let newTx = {

                id: editingTxId || Date.now(),

                type, amount, from: fromAcc.name,

                category: type === 'transfer' ? 'โอนเงิน' : (document.getElementById('tx-category')?.value || 'อื่นๆ'),

                note: document.getElementById('tx-note')?.value || '',

                date: document.getElementById('tx-date')?.value || new Date().toISOString().split('T')[0]

            };


            if (type === 'transfer') {

                const toId = parseInt(document.getElementById('to-account')?.value);

                const toAcc = accounts.find(a => a.id === toId);

                if (toAcc) {

                    fromAcc.balance -= amount;

                    toAcc.balance += amount;

                    newTx.to = toAcc.name;

                }

            } else if (type === 'expense') {

                fromAcc.balance -= amount;

            } else {

                fromAcc.balance += amount;

            }


            if (editingTxId) {

                const idx = transactions.findIndex(t => t.id === editingTxId);

                if(idx !== -1) transactions[idx] = newTx;

            } else transactions.unshift(newTx);


            resetForm(); renderAll();

        }


        function revertBalance(id) {

            const tx = transactions.find(t => t.id === id);

            if (!tx) return;

            const f = accounts.find(a => a.name === tx.from);

            const t = accounts.find(a => a.name === tx.to);

            if (tx.type === 'expense') { if(f) f.balance += tx.amount; }

            else if (tx.type === 'income') { if(f) f.balance -= tx.amount; }

            else { if(f) f.balance += tx.amount; if(t) t.balance -= tx.amount; }

        }


        function editTx(id) {

            const tx = transactions.find(t => t.id === id);

            if(!tx) return;

            editingTxId = id;

            switchTab('overview');

            document.getElementById('form-container')?.classList.add('edit-mode');

            document.getElementById('edit-indicator')?.classList.remove('hidden');

            setTxType(tx.type);

            if(document.getElementById('tx-amount')) document.getElementById('tx-amount').value = tx.amount;

            if(document.getElementById('tx-note')) document.getElementById('tx-note').value = tx.note;

            if(document.getElementById('tx-date')) document.getElementById('tx-date').value = tx.date;

            if(document.getElementById('tx-category')) document.getElementById('tx-category').value = tx.category;

            const submitBtn = document.getElementById('submit-btn');

            if(submitBtn) submitBtn.innerText = "อัปเดตรายการ";

            document.getElementById('cancel-edit-btn')?.classList.remove('hidden');

        }

        function renderBudgetBars() {

            const container = document.getElementById('budget-bars');

            if(!container) return;

            const cur = new Date().getMonth();

            const b = annualBudgets[cur] || {};

            const totals = {};

            transactions.filter(t => new Date(t.date).getMonth() === cur).forEach(t => totals[t.category] = (totals[t.category]||0) + t.amount);

            

            container.innerHTML = CATEGORIES.slice(0, 4).map(c => {

                const limit = b[c.name] || 0;

                const actual = totals[c.name] || 0;

                const pct = limit > 0 ? Math.min((actual/limit)*100, 100) : 0;

                return `

                    <div class="space-y-1">

                        <div class="flex justify-between text-[9px] font-black uppercase">

                            <span class="text-slate-500">${c.name}</span>

                            <span>฿${actual.toLocaleString()} / ฿${limit.toLocaleString()}</span>

                        </div>

                        <div class="h-1.5 bg-slate-100 rounded-full overflow-hidden">

                            <div class="h-full bg-brand-500 transition-all duration-700" style="width: ${pct}%"></div>

                        </div>

                    </div>

                `;

            }).join('');

        }

        function updateCalculations() {

            const assets = accounts.filter(a => a.type !== 'debt').reduce((s, a) => s + a.balance, 0);

            const debt = accounts.filter(a => a.type === 'debt').reduce((s, a) => s + a.balance, 0);

            const net = assets - debt;

            const disp = document.getElementById('display-net-worth');

            if(disp) {

                disp.innerText = `฿ ${net.toLocaleString(undefined, {minimumFractionDigits: 2})}`;

                disp.style.color = net < 0 ? '#f43f5e' : '#047857';

            }

        }
